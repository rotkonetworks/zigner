44/// Get display address for a multisigner - checks for stored addresses first
45fn get_display_address(
46    database: &sled::Db,
47    multisigner: &MultiSigner,
48    optional_prefix: Option<u16>,
49    encryption: Encryption,
50    network_name: Option<&str>,
51) -> String {
52    // For Penumbra, try to get the stored bech32m address
53    // Never fall back to another encoding: a key added before zigner#35 shows
54    // no address until it is derived again with the seed.
55    if encryption == Encryption::Penumbra {
56        let key_id_hex = hex::encode(multisigner_to_public(multisigner));
57        return crate::penumbra::get_penumbra_address(database, &key_id_hex)
58            .ok()
59            .flatten()
60            .unwrap_or_default();
61    }
62    // For Zcash, try to get the stored unified address (u1...)
63    #[cfg(feature = "zcash")]
64    if encryption == Encryption::Zcash {
65        let pubkey_hex = hex::encode(multisigner_to_public(multisigner));
66        if let Ok(Some(address)) = crate::zcash::get_zcash_address(database, &pubkey_hex) {
67            return address;
68        }
69    }
70    // For Cosmos, compute bech32 address on-the-fly from pubkey + chain prefix
71    #[cfg(feature = "cosmos")]
72    if encryption == Encryption::Cosmos {
73        let pubkey_bytes = multisigner_to_public(multisigner);
74        if let Ok(address) =
75            crate::cosmos::pubkey_to_bech32_address(&pubkey_bytes, network_name.unwrap_or("cosmos"))
76        {
77            return address;
78        }
79    }
80    // Fall back to default display
81    print_multisigner_as_base58_or_eth_address(multisigner, optional_prefix, encryption)
82}

84/// Return a `Vec` with all seed names with seed key identicons if seed key is
85/// available.
86///
87/// Function processes all seeds known to Vault KMS (which are input as
88/// `&[String]`), including seeds without any corresponding addresses currently
89/// known to Vault (orphans).
90///
91/// If the same seed has more than one seed key in the database, i.e. it has
92/// been used to create seed keys with more than one
93/// [`Encryption`] algorithm, only one
94/// identicon is selected, in order of preference: `Sr25519`, `Ed25519`,
95/// `Ecdsa`.
96pub fn get_all_seed_names_with_identicons(
97    database: &sled::Db,
98    names_phone_knows: &[String],
99) -> Result<Vec<SeedNameCard>> {
100    let mut data_set: HashMap<String, Identicon> = HashMap::new();
101    let mut derivation_count: HashMap<String, u32> = HashMap::new();
102    let mut used_in_networks: HashMap<String, HashSet<String>> = HashMap::new();
103    let mut network_names_cache: HashMap<NetworkSpecsKey, String> = HashMap::new();
104
105    for (multisigner, address_details) in get_all_addresses(database)?.into_iter() {
106        if address_details.is_root() {
107            let identicon =
108                make_identicon_from_multisigner(&multisigner, address_details.identicon_style());
109            data_set.insert(address_details.seed_name.to_string(), identicon);
110        } else {
111            if let Some(network) = address_details.network_id {
112                if !network_names_cache.contains_key(&network) {
113                    let name = get_network_specs(database, &network)?.specs.name;
114                    network_names_cache.insert(network.clone(), name);
115                }
116                if let Some(name) = network_names_cache.get(&network) {
117                    used_in_networks
118                        .entry(address_details.seed_name.to_string())
119                        .or_default()
120                        .insert(name.to_string());
121                }
122            }
123
124            derivation_count
125                .entry(address_details.seed_name.to_string())
126                .and_modify(|e| *e += 1)
127                .or_insert(1);
128            if !data_set.contains_key(&address_details.seed_name) {
129                data_set.insert(address_details.seed_name.to_string(), Identicon::default());
130            }
131        }
132    }
133    for x in names_phone_knows.iter() {
134        if !data_set.contains_key(x) {
135            data_set.insert(x.to_string(), Identicon::default());
136        }
137    }
138    let mut res: Vec<_> = data_set
139        .into_iter()
140        .map(|(seed_name, identicon)| {
141            let mut used_in_networks = used_in_networks
142                .get(&seed_name)
143                .cloned()
144                .map(|hs| hs.into_iter().collect())
145                .unwrap_or_else(Vec::new);
146            used_in_networks.sort();
147            SeedNameCard {
148                seed_name: seed_name.clone(),
149                identicon,
150                derived_keys_count: *derivation_count.get(&seed_name).unwrap_or(&0),
151                used_in_networks,
152            }
153        })
154        .collect();
155    res.sort_by(|a, b| a.seed_name.cmp(&b.seed_name));
156    Ok(res)
157}

159/// Return a `Vec` with address-associated public data for all addresses from the
160/// Vault database.
161///
162/// Function is used to show users all possible addresses, when selecting the
163/// address to generate
164/// `SufficientCrypto` for signing
165/// updates with the Vault.
166pub fn print_all_identities(database: &sled::Db) -> Result<Vec<MRawKey>> {
167    Ok(get_all_addresses(database)?
168        .into_iter()
169        .filter_map(|(multisigner, address_details)| {
170            match &address_details.network_id {
171                Some(id) => {
172                    let network_specs = get_network_specs(database, id).unwrap();
173
174                    let address_key = AddressKey::new(
175                        multisigner.clone(),
176                        Some(network_specs.specs.genesis_hash),
177                    ); // to click
178                    let public_key = multisigner_to_public(&multisigner); // to display
179                    let style = address_details.identicon_style();
180                    let identicon = make_identicon_from_multisigner(&multisigner, style);
181                    Some(MRawKey {
182                        address: Address {
183                            identicon,
184                            has_pwd: address_details.has_pwd,
185                            path: address_details.path,
186                            secret_exposed: address_details.secret_exposed,
187                            seed_name: address_details.seed_name,
188                        },
189                        address_key: hex::encode(address_key.key()),
190                        public_key: hex::encode(public_key),
191                        network_logo: network_specs.specs.logo,
192                    })
193                }
194                None => None,
195            }
196        })
197        .collect())
198}

200pub fn keys_by_seed_name(database: &sled::Db, seed_name: &str) -> Result<MKeysNew> {
201    let (root, derived): (Vec<_>, Vec<_>) = get_addresses_by_seed_name(database, seed_name)?
202        .into_iter()
203        .partition(|(_, address)| address.is_root());
204
205    let root = root.first().map(|root| {
206        let address = Address {
207            has_pwd: false,
208            path: String::new(),
209            seed_name: seed_name.to_string(),
210            identicon: make_identicon_from_multisigner(&root.0, root.1.identicon_style()),
211            secret_exposed: root.1.secret_exposed,
212        };
213        // TODO: root always prefix 42 for substrate.
214        let address_key = hex::encode(AddressKey::new(root.0.clone(), None).key());
215        MAddressCard {
216            base58: get_display_address(database, &root.0, None, root.1.encryption, None),
217            address_key,
218            address,
219        }
220    });
221
222    let mut set = vec![];
223    for (multisigner, address_details) in derived.into_iter() {
224        if let Some(id) = &address_details.network_id {
225            let network_specs = get_network_specs(database, id)?;
226
227            let identicon =
228                make_identicon_from_multisigner(&multisigner, address_details.identicon_style());
229            let base58 = get_display_address(
230                database,
231                &multisigner,
232                Some(network_specs.specs.base58prefix),
233                network_specs.specs.encryption,
234                Some(&network_specs.specs.name),
235            );
236            let address_key = hex::encode(
237                AddressKey::new(multisigner.clone(), Some(network_specs.specs.genesis_hash)).key(),
238            );
239            let address = Address {
240                path: address_details.path,
241                has_pwd: address_details.has_pwd,
242                identicon,
243                secret_exposed: address_details.secret_exposed,
244                seed_name: seed_name.to_owned(),
245            };
246            let key = MKeysCard {
247                address,
248                base58,
249                address_key,
250                swiped: false,
251            };
252            let network_specs_key = NetworkSpecsKey::from_parts(
253                &network_specs.specs.genesis_hash,
254                &network_specs.specs.encryption,
255            );
256            let network = MSCNetworkInfo {
257                network_title: network_specs.specs.name,
258                network_logo: network_specs.specs.logo,
259                network_specs_key: hex::encode(network_specs_key.key()),
260            };
261
262            set.push(MKeyAndNetworkCard { key, network })
263        }
264    }
265
266    Ok(MKeysNew { root, set })
267}

284/// Return `Vec` with network information for all networks in the Vault database,

286pub fn show_all_networks_with_flag(
287    database: &sled::Db,
288    network_specs_key: &NetworkSpecsKey,
289) -> Result<MNetworkMenu> {

301
302    Ok(MNetworkMenu { networks })

339/// Prepare export key screen struct [`MKeyDetails`].
340///
341/// For QR code the address information is put in format
342/// `substrate:{address as base58}:0x{network genesis hash}`
343/// `ethereum:{address as hex}:0x{network genesis hash}`
344/// `cosmos:`, `zcash:` and `penumbra:` wrap the chain's own address string.
345/// transformed into bytes, to be compatible with `polkadot-js` interface.
346/// Note that no [`Encryption`] algorithm
347/// information is contained in the QR code. If there are multiple `Encryption`
348/// algorithms supported by the network, the only visible difference in exports
349/// would be the identicon.
350pub fn export_key(
351    database: &sled::Db,
352    multisigner: &MultiSigner,
353    expected_seed_name: &str,
354    network_specs_key: &NetworkSpecsKey,
355) -> Result<MKeyDetails> {
356    let ordered_network_specs = get_network_specs(database, network_specs_key)?;
357    let network_specs = ordered_network_specs.specs;
358    let address_key = AddressKey::new(multisigner.clone(), Some(network_specs.genesis_hash));
359    let address_details = get_address_details(database, &address_key);
360    let address_details = if address_details.is_err() {
361        let address_key = AddressKey::new(multisigner.clone(), None);
362        get_address_details(database, &address_key)
363    } else {
364        address_details
365    };
366
367    let address_details = address_details?;
368
369    if address_details.seed_name != expected_seed_name {
370        return Err(Error::SeedNameNotMatching {
371            address_key,
372            expected_seed_name: expected_seed_name.to_string(),
373            real_seed_name: address_details.seed_name,
374        });
375    }
376    let base58 = get_display_address(
377        database,
378        multisigner,
379        Some(network_specs.base58prefix),
380        network_specs.encryption,
381        Some(&network_specs.name),
382    );
383
384    let public_key = multisigner_to_public(multisigner);
385    // a Penumbra key with no address yet gets no address QR
386    let no_address = network_specs.encryption == Encryption::Penumbra && base58.is_empty();
387    let identicon = make_identicon_from_multisigner(multisigner, address_details.identicon_style());
388    let qr = {
389        if address_details.network_id.as_ref() == Some(network_specs_key) {
390            let prefix = if network_specs.encryption == Encryption::Ethereum {
391                "ethereum"
392            } else if network_specs.encryption == Encryption::Cosmos {
393                "cosmos"
394            } else if network_specs.encryption == Encryption::Zcash {
395                "zcash"
396            } else if network_specs.encryption == Encryption::Penumbra {
397                "penumbra"
398            } else {
399                "substrate"
400            };
401
402            QrData::Regular {
403                data: if no_address {
404                    Vec::new()
405                } else {
406                    format!(
407                        "{}:{}:0x{}",
408                        prefix,
409                        base58,
410                        hex::encode(network_specs.genesis_hash)
411                    )
412                    .as_bytes()
413                    .to_vec()
414                },
415            }
416        } else {
417            return Err(Error::NetworkSpecsKeyForAddressNotFound {
418                network_specs_key: network_specs_key.to_owned(),
419                address_key,
420            });
421        }
422    };
423    let address = Address {
424        path: address_details.path,
425        has_pwd: address_details.has_pwd,
426        identicon,
427        seed_name: address_details.seed_name,
428        secret_exposed: address_details.secret_exposed,
429    };
430
431    let network_info = MSCNetworkInfo {
432        network_title: network_specs.name,
433        network_logo: network_specs.logo,
434        network_specs_key: hex::encode(network_specs_key.key()),
435    };
436
437    Ok(MKeyDetails {
438        qr,
439        pubkey: hex::encode(public_key),
440        network_info,
441        base58,
442        address,
443    })
444}

446/// Prepare seed backup screen struct [`MBackup`] for given seed name.
447///
448/// Function inputs seed name, outputs `Vec` with all known derivations in all
449/// networks.
450pub fn backup_prep(database: &sled::Db, seed_name: &str) -> Result<MBackup> {
451    let networks = get_all_networks(database)?;
452    if networks.is_empty() {
453        return Err(Error::NoNetworksAvailable);
454    }
455    let mut derivations = Vec::new();
456    for network in networks.into_iter() {
457        let id_set: Vec<_> = addresses_set_seed_name_network(
458            database,
459            seed_name,
460            &NetworkSpecsKey::from_parts(&network.specs.genesis_hash, &network.specs.encryption),
461        )?
462        .into_iter()
463        .map(|a| DerivationEntry {
464            path: a.1.path,
465            has_pwd: a.1.has_pwd,
466        })
467        .collect();
468        if !id_set.is_empty() {
469            derivations.push(DerivationPack {
470                network_title: network.specs.name,
471                network_logo: network.specs.logo,
472                network_order: network.order.to_string(),
473                id_set,
474            });
475        }
476    }
477
478    derivations.sort_by(|a, b| a.network_order.cmp(&b.network_order));
479
480    Ok(MBackup {
481        seed_name: seed_name.to_string(),
482        derivations,
483    })
484}

510/// Return [`NavDerivationCheck`] with allowed action details for new key derivation.
511///
512/// Function is used to dynamically check from the frontend if user is allowed
513/// to proceed with the proposed derived key generation.
514///
515/// User is allowed to try to proceed only if the derivation is valid and, in
516/// case of derivations without password, if the derivation does not already
517/// exist in the database. Passworded valid derivations are allowed to proceed,
518/// but result in an error later on, if the derivation exists.
519///
520/// Function makes only preliminary check on password-free derivations, it
521/// **does not** use seed phrase and does not calculate the [`AddressKey`], i.e.
522/// it can't check passworded derivations, and allows them to proceed anyway.
523pub fn dynamic_path_check(
524    database: &sled::Db,
525    seed_name: &str,
526    path: &str,
527    network_specs_key_hex: &str,
528) -> NavDerivationCheck {
529    match NetworkSpecsKey::from_hex(network_specs_key_hex) {
530        Ok(key) => dynamic_path_check_unhexed(database, seed_name, path, &key),
531        Err(e) => NavDerivationCheck {
532            error: Some(e.to_string()),
533            ..Default::default()
534        },
535    }
536}

538fn dynamic_path_check_unhexed(
539    database: &sled::Db,
540    seed_name: &str,
541    path: &str,
542    network_specs_key: &NetworkSpecsKey,
543) -> NavDerivationCheck {
544    match get_network_specs(database, network_specs_key) {
545        Ok(ordered_network_specs) => {
546            match derivation_check(database, seed_name, path, network_specs_key) {
547                Ok(DerivationCheck::BadFormat) => NavDerivationCheck {
548                    button_good: false,
549                    ..Default::default()
550                },
551                Ok(DerivationCheck::Password) => NavDerivationCheck {
552                    button_good: true,
553                    where_to: Some(DerivationDestination::Pwd),
554                    ..Default::default()
555                },
556                Ok(DerivationCheck::NoPassword(None)) => NavDerivationCheck {
557                    button_good: true,
558                    where_to: Some(DerivationDestination::Pin),
559                    ..Default::default()
560                },
561                Ok(DerivationCheck::NoPassword(Some((multisigner, address_details)))) => {
562                    let address_base58 = get_display_address(
563                        database,
564                        &multisigner,
565                        Some(ordered_network_specs.specs.base58prefix),
566                        address_details.encryption,
567                        Some(&ordered_network_specs.specs.name),
568                    );
569                    let identicon = make_identicon_from_multisigner(
570                        &multisigner,
571                        address_details.identicon_style(),
572                    );
573                    let address_key = hex::encode(
574                        AddressKey::new(
575                            multisigner,
576                            Some(ordered_network_specs.specs.genesis_hash),
577                        )
578                        .key(),
579                    );
580                    let collision_display = MAddressCard {
581                        base58: address_base58,
582                        address_key,
583                        address: Address {
584                            path: address_details.path,
585                            has_pwd: address_details.has_pwd,
586                            identicon,
587                            seed_name: seed_name.to_string(),
588                            secret_exposed: address_details.secret_exposed,
589                        },
590                    };
591                    NavDerivationCheck {
592                        button_good: false,
593                        collision: Some(collision_display),
594                        ..Default::default()
595                    }
596                }
597                Err(e) => NavDerivationCheck {
598                    error: Some(e.to_string()),
599                    ..Default::default()
600                },
601            }
602        }
603        Err(e) => NavDerivationCheck {
604            error: Some(e.to_string()),
605            ..Default::default()
606        },
607    }
608}

610/// Return [`MNetworkDetails`] with network specs and metadata set information
611/// for network with given [`NetworkSpecsKey`].
612pub fn network_details_by_key(
613    database: &sled::Db,
614    network_specs_key: &NetworkSpecsKey,
615) -> Result<MNetworkDetails> {
616    let OrderedNetworkSpecs {
617        specs:
618            NetworkSpecs {
619                base58prefix,
620                color,
621                decimals,
622                encryption,
623                genesis_hash,
624                logo,
625                name,
626                path_id,
627                secondary_color,
628                title,
629                unit,
630            },
631        order,
632    } = get_network_specs(database, network_specs_key)?;
633    let verifier_key = VerifierKey::from_parts(genesis_hash);
634    let general_verifier = get_general_verifier(database)?;
635    let current_verifier = get_valid_current_verifier(database, &verifier_key)?;
636    let meta: Vec<_> = get_meta_values_by_name(database, &name)?
637        .into_iter()
638        .map(|m| {
639            let meta_hash = blake2_256(&m.meta);
640            let meta_id_pic = pic_meta(&meta_hash);
641
642            MMetadataRecord {
643                specname: m.name,
644                specs_version: m.version.to_string(),
645                meta_hash: hex::encode(meta_hash),
646                meta_id_pic,
647            }
648        })
649        .collect();
650
651    let (ttype, details) = match current_verifier {
652        ValidCurrentVerifier::General => ("general".to_string(), general_verifier.show_card()),
653        ValidCurrentVerifier::Custom { v } => ("custom".to_string(), v.show_card()),
654    };
655    let current_verifier = MVerifier { ttype, details };
656
657    Ok(MNetworkDetails {
658        base58prefix,
659        color,
660        decimals,
661        encryption,
662        genesis_hash,
663        logo,
664        name,
665        order: order.to_string(),
666        path_id,
667        secondary_color,
668        title,
669        unit,
670        current_verifier,
671        meta,
672    })
673}

764/// Get possible options of English `bip39` words that start with user-entered
765/// word part.
766///
767/// List length limit is [`MAX_WORDS_DISPLAY`].
768pub fn guess(word_part: &str) -> Vec<&'static str> {
769    let dictionary = Language::English.wordlist();
770    let words = dictionary.get_words_by_prefix(word_part);
771    if words.len() > MAX_WORDS_DISPLAY {
772        words[..MAX_WORDS_DISPLAY].to_vec()
773    } else {
774        words.to_vec()
775    }
776}

778/// Maximum word count in `bip39` standard.
779///
780/// See <https://docs.rs/tiny-bip39/0.8.2/src/bip39/mnemonic_type.rs.html#60>
781pub const BIP_CAP: usize = 24;
782
783/// Maximum word length in `bip39` standard.
784pub const WORD_LENGTH: usize = 8;
785
786/// Zeroizeable seed phrase draft.
787#[derive(Debug, Clone, Zeroize, ZeroizeOnDrop)]
788pub struct SeedDraft {
789    /// User-entered word part.
790    user_input: String,
791
792    /// Already completed `bip39` words.
793    saved: Vec<SeedElement>,
794}
795
796/// Zeroizeable wrapper around complete `bip39` word entered by user.
797#[derive(Debug, Clone, Zeroize, ZeroizeOnDrop)]
798struct SeedElement(String);
799
800impl SeedElement {
801    /// Make `SeedElement` from checked `bip39` word.
802    fn from_checked_str(word: &str) -> Self {
803        let mut new = String::with_capacity(WORD_LENGTH);
804        new.push_str(word);
805        SeedElement(new)
806    }
807
808    /// Get `bip39` word from the `SeedElement`.
809    fn word(&self) -> &str {
810        &self.0
811    }
812}
813
814impl SeedDraft {
815    /// Start new `SeedDraft`
816    pub fn initiate() -> Self {
817        Self {
818            user_input: String::with_capacity(WORD_LENGTH), // capacity corresponds to maximum word length in `bip39` standard;
819            saved: Vec::with_capacity(BIP_CAP), // capacity corresponds to maximum word count in `bip39` standard; set here to avoid reallocation;
820        }
821    }
822
823    /// Modify `SeedDraft` with updated `user_text` from the frontend.
824    ///
825    /// Note that `user_text` input by default starts with ' ' (space). If user
826    /// removes this space, it results in removing whole previous word.
827    pub fn text_field_update(&mut self, user_text: &str) {
828        if self.saved.len() <= BIP_CAP {
829            if user_text.is_empty() {
830                // user has removed all text, including the first default symbol
831                // if there are words in draft, remove the last one
832                self.remove_last();
833                // restore the user input to empty one
834                self.user_input.clear();
835            } else {
836                let user_text = user_text.trim_start();
837
838                // ' ' (space) in the end of the word indicates user attempt to
839                // submit the word into seed phrase
840                if user_text.ends_with(' ') {
841                    let word = user_text.trim();
842                    if self.added(word, None) {
843                        self.user_input.clear() // added the word successfully, clear `user_input`
844                    } else if !guess(word).is_empty() {
845                        self.user_input = String::from(word) // did not add the word, there are still possible variants, keep trimmed `user_input`
846                    }
847                } else if !guess(user_text).is_empty() {
848                    self.user_input = String::from(user_text)
849                }
850            }
851        } else {
852            self.user_input.clear()
853        }
854    }
855
856    /// User tries to add the word to the `saved` field of the `SeedDraft`.
857    /// Output is `true` if addition happens. `SeedDraft` gets modified in the
858    /// process.
859    ///
860    /// Optional `position` input could be used to mark the position in seed
861    /// phrase to add the word to.
862    pub fn added(&mut self, word: &str, position: Option<u32>) -> bool {
863        // maximum number of the words is not reached
864        if self.saved.len() < BIP_CAP {
865            let guesses = guess(word);
866            let definitive_guess = {
867                if guesses.len() == 1 {
868                    Some(guesses[0]) // only one possible variant
869                } else if guesses.contains(&word) {
870                    Some(word) // exactly matching variant
871                } else {
872                    None // no definitive match, no addition
873                }
874            };
875            if let Some(guess) = definitive_guess {
876                let new = SeedElement::from_checked_str(guess);
877                match position {
878                    Some(p) => {
879                        let p = p as usize;
880                        if p <= self.saved.len() {
881                            self.saved.insert(p, new) // position is reasonable, use it
882                        } else {
883                            self.saved.push(new) // position is **not** reasonable, add word at the end of the list
884                        }
885                    }
886                    None => self.saved.push(new), // no position, add word at the end of the list
887                }
888                self.user_input.clear();
889                true
890            } else {
891                false
892            }
893        } else {
894            false
895        }
896    }
897
898    /// Remove word at given position from the saved seed phrase draft.
899    pub fn remove(&mut self, position: u32) {
900        let position = position as usize;
901        if position < self.saved.len() {
902            self.saved.remove(position);
903        }
904    }
905
906    /// Remove last word from the saved seed phrase draft.
907    pub fn remove_last(&mut self) {
908        if !self.saved.is_empty() {
909            self.saved.remove(self.saved.len() - 1);
910        }
911    }
912
913    /// Make a `String` with a seed phrase draft.
914    ///
915    /// Returned string contains a secret and should be handled as such.
916    pub fn draft(&self) -> Vec<String> {
917        self.saved.iter().map(|w| w.word().to_string()).collect()
918    }
919
920    /// Combines all draft elements into seed phrase proposal,
921    /// and checks its validity.
922    /// If valid, outputs secret seed phrase.
923    pub fn try_finalize(&self) -> Option<String> {
924        let mut seed_phrase_proposal = String::with_capacity((WORD_LENGTH + 1) * BIP_CAP);
925        for (i, x) in self.saved.iter().enumerate() {
926            if i > 0 {
927                seed_phrase_proposal.push(' ');
928            }
929            seed_phrase_proposal.push_str(x.word());
930        }
931        if Mnemonic::validate(&seed_phrase_proposal, Language::English).is_ok() {
932            Some(seed_phrase_proposal)
933        } else {
934            seed_phrase_proposal.zeroize();
935            None
936        }
937    }
938
939    /// Output the user input back into user interface.
940    pub fn user_input(&self) -> &str {
941        &self.user_input
942    }
943}
