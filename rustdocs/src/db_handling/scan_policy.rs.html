1//! What the camera may hand to a parser on this device.
2//!
3//! The camera is the only untrusted input. The camera layer classifies each
4//! code, and a network's content parser is reachable only when this device
5//! holds something it would be used with: no Penumbra key, no Penumbra
6//! parsing; no Zcash key or FROST wallet, no Zcash parsing. The answer is
7//! derived from the database every time it is asked, so there is no setting
8//! that can drift from what the device holds.
9//!
10//! This gates content parsers only. The multi-frame transport decoder
11//! (`qr_reader_phone`: RaptorQ, legacy frames, banana-split shares) runs on
12//! byte-mode codes before classification and is not covered here.
13//!
14//! Three kinds stay open on an empty device, because they are how a device
15//! stops being empty: recovery (banana split and backup import), creating
16//! a FROST wallet, and signed module updates.
17
18use constants::FROST_KEYS_TREE;
19use definitions::crypto::Encryption;
20
21use crate::identities::get_all_addresses;
22use crate::Result;
23
24/// Signed protocol-module updates. The signature is checked before use.
25pub const MODULE: &str = "module";
26/// Seed recovery and backup import.
27pub const BACKUP: &str = "backup";
28/// FROST key generation, which starts before any FROST wallet exists.
29pub const FROST_DKG: &str = "frost-dkg";
30/// FROST signing rounds.
31pub const FROST: &str = "frost";
32/// Zcash notes and PCZT signing requests.
33pub const ZCASH: &str = "zcash";
34/// Penumbra signing requests.
35pub const PENUMBRA: &str = "penumbra";
36/// ZID identity challenges, derived from a seed.
37pub const AUTH: &str = "auth";
38
39/// The kinds of QR code this device's camera may parse. Polkadot and
40/// Cosmos are never in it.
41pub fn readable(database: &sled::Db) -> Result<Vec<&'static str>> {
42    // A FROST wallet is any entry in its tree; nothing is deserialized.
43    let has_frost_wallet = !database.open_tree(FROST_KEYS_TREE)?.is_empty();
44    let (mut any_key, mut zcash_key, mut penumbra_key) = (false, false, false);
45    for (_, address_details) in get_all_addresses(database)? {
46        any_key = true;
47        match address_details.encryption {
48            Encryption::Zcash => zcash_key = true,
49            Encryption::Penumbra => penumbra_key = true,
50            _ => {}
51        }
52    }
53
54    let mut out = vec![MODULE, BACKUP, FROST_DKG];
55    if has_frost_wallet {
56        out.push(FROST);
57    }
58    if has_frost_wallet || zcash_key {
59        out.push(ZCASH);
60    }
61    if penumbra_key {
62        out.push(PENUMBRA);
63    }
64    if any_key {
65        out.push(AUTH);
66    }
67    Ok(out)
68}
