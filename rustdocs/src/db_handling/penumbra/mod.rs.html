1//! Penumbra keys as Zigner stores them
2//!
3//! Spend keys, viewing keys and addresses come from penumbra-sdk, the same
4//! derivation the FVK export and signing use: `SpendKey::from_seed_phrase_bip44`
5//! at `m/44'/6532'/account'`. The receive address is index 0 of that key, which
6//! is the address Zafu shows after importing the exported FVK.
7//!
8//! [`key_id`] is only a database identifier. Zigner 0.12.0 and older keyed
9//! Penumbra entries by a verification key derived from a blake2b hash of the
10//! seed, and existing installs keep those keys, so new entries use the same id.
11//! It is not a Penumbra key and must not be shown as one.
12
13use crate::error::Result;
14use constants::{PENUMBRA_ADDRS_UNSPENDABLE, PENUMBRA_RECEIVE_ADDRS};
15
16/// Database id of the Penumbra key for `account`, unchanged since Zigner added
17/// Penumbra so stored entries keep matching.
18#[cfg(feature = "penumbra")]
19pub fn key_id(seed_phrase: &str, account: u32) -> Result<[u8; 32]> {
20    use bip39::{Language, Mnemonic, Seed};
21    use penumbra_keys::keys::{SpendKey, SpendKeyBytes};
22    use zeroize::Zeroize;
23
24    let mnemonic = Mnemonic::from_phrase(seed_phrase, Language::English)
25        .map_err(|e| crate::error::Error::Other(anyhow::anyhow!("Invalid mnemonic: {e}")))?;
26    let seed = Seed::new(&mnemonic, "");
27    let hash = blake2b_simd::Params::new()
28        .personal(b"PenumbraSpendKey")
29        .key(seed.as_bytes())
30        .hash(format!("m/44'/6532'/{account}'/0/0").as_bytes());
31    let mut bytes = [0u8; 32];
32    bytes.copy_from_slice(&hash.as_bytes()[..32]);
33    let key: SpendKey = SpendKeyBytes(bytes).into();
34    bytes.zeroize();
35    Ok(key.full_viewing_key().spend_verification_key().to_bytes())
36}

38/// The receive address for `account`: index 0 of the BIP44 account's key.
39#[cfg(feature = "penumbra")]
40pub fn receive_address(seed_phrase: &str, account: u32) -> Result<String> {
41    use penumbra_keys::keys::{AddressIndex, Bip44Path, SeedPhrase, SpendKey};
42    use std::str::FromStr;
43
44    let phrase = SeedPhrase::from_str(seed_phrase)
45        .map_err(|e| crate::error::Error::Other(anyhow::anyhow!("Invalid mnemonic: {e}")))?;
46    let key = SpendKey::from_seed_phrase_bip44(phrase, &Bip44Path::new(account));
47    let (address, _) = key.full_viewing_key().payment_address(AddressIndex::new(0));
48    Ok(address.to_string())
49}

51/// Store the receive address for a key id. Drops the pre-fix address tree.
52pub fn store_penumbra_address(
53    database: &sled::Db,
54    key_id_hex: &str,
55    bech32m_address: &str,
56) -> Result<()> {
57    database.drop_tree(PENUMBRA_ADDRS_UNSPENDABLE)?;
58    let tree = database.open_tree(PENUMBRA_RECEIVE_ADDRS)?;
59    tree.insert(key_id_hex.as_bytes(), bech32m_address.as_bytes())?;
60    Ok(())
61}

63/// Every stored receive address on this device. The review screen marks an
64/// output as the user's own only when it pays one of these.
65pub fn receive_addresses(database: &sled::Db) -> Result<Vec<String>> {
66    let tree = database.open_tree(PENUMBRA_RECEIVE_ADDRS)?;
67    let mut out = Vec::new();
68    for entry in tree.iter() {
69        let (_, bytes) = entry?;
70        if let Ok(address) = String::from_utf8(bytes.to_vec()) {
71            out.push(address);
72        }
73    }
74    Ok(out)
75}

77/// The stored receive address for a key id. `None` for a key added before the
78/// fix whose address has not been derived again yet.
79pub fn get_penumbra_address(database: &sled::Db, key_id_hex: &str) -> Result<Option<String>> {
80    let tree = database.open_tree(PENUMBRA_RECEIVE_ADDRS)?;
81    match tree.get(key_id_hex.as_bytes())? {
82        Some(bytes) => {
83            let address = String::from_utf8(bytes.to_vec())
84                .map_err(|e| crate::error::Error::Other(anyhow::anyhow!("Invalid UTF-8: {}", e)))?;
85            Ok(Some(address))
86        }
87        None => Ok(None),
88    }
89}

91/// Derive the receive address for `account` and store it under its key id.
92/// Run whenever the seed is at hand, so keys added before the fix get theirs.
93#[cfg(feature = "penumbra")]
94pub fn refresh_receive_address(
95    database: &sled::Db,
96    seed_phrase: &str,
97    account: u32,
98) -> Result<String> {
99    let address = receive_address(seed_phrase, account)?;
100    let id = hex::encode(key_id(seed_phrase, account)?);
101    store_penumbra_address(database, &id, &address)?;
102    Ok(address)
103}
