1//! zcash transaction parsing
2//!
3//! parses zcash sign requests from QR codes and prepares them for signing.
4//!
5//! QR format (from zafu wallet):
6//! ```text
7//! [0x53][0x04][0x02]           - prelude
8//! [flags: 1 byte]              - bit 0: mainnet
9//! [account_index: 4 bytes LE]
10//! [sighash: 32 bytes]
11//! [action_count: 2 bytes LE]
12//! [alphas: 32 bytes each]
13//! [summary_len: 2 bytes LE]
14//! [summary: summary_len bytes]
15//! ```
16

25/// zcash chain identifier in QR prelude (0x04)
26pub const ZCASH_CRYPTO_TYPE: u8 = 0x04;
27
28/// zcash sign request type (0x02)
29pub const ZCASH_SIGN_REQUEST_TYPE: u8 = 0x02;
30
31/// parsed zcash sign request
32#[derive(Debug, Clone)]
33pub struct ZcashSignRequestData {
34    /// account index for key derivation
35    pub account_index: u32,
36    /// the transaction sighash (32 bytes)
37    pub sighash: [u8; 32],
38    /// orchard action randomizers (alpha values)
39    pub orchard_alphas: Vec<[u8; 32]>,
40    /// human-readable summary for display
41    pub summary: String,
42    /// network: true = mainnet, false = testnet
43    pub mainnet: bool,
44    /// true if this is a shielding (transparent → orchard) transaction
45    pub shielding: bool,
46}
47
48impl ZcashSignRequestData {
49    /// parse from QR hex string
50    pub fn from_qr_hex(hex_str: &str) -> Result<Self> {
51        let data = unhex(hex_str)?;
52        Self::from_qr_bytes(&data)
53    }
54
55    /// parse from QR bytes
56    pub fn from_qr_bytes(data: &[u8]) -> Result<Self> {
57        // validate prelude: [0x53][0x04][0x02]
58        if data.len() < 42 {
59            // minimum: 3 + 1 + 4 + 32 + 2 = 42
60            return Err(Error::ZcashParseError("QR data too short".to_string()));
61        }
62        if data[0] != 0x53 {
63            return Err(Error::NotSubstrate(format!("{:02x}", data[0])));
64        }
65        if data[1] != ZCASH_CRYPTO_TYPE {
66            return Err(Error::ZcashParseError(format!(
67                "expected crypto type 0x04, got 0x{:02x}",
68                data[1]
69            )));
70        }
71        if data[2] != ZCASH_SIGN_REQUEST_TYPE {
72            return Err(Error::ZcashParseError(format!(
73                "expected tx type 0x02, got 0x{:02x}",
74                data[2]
75            )));
76        }
77
78        let mut offset = 3;
79
80        // parse flags
81        let flags = data[offset];
82        let mainnet = (flags & 0x01) != 0;
83        let shielding = (flags & 0x02) != 0;
84        offset += 1;
85
86        // account index
87        if offset + 4 > data.len() {
88            return Err(Error::ZcashParseError(
89                "account index truncated".to_string(),
90            ));
91        }
92        let account_index = u32::from_le_bytes([
93            data[offset],
94            data[offset + 1],
95            data[offset + 2],
96            data[offset + 3],
97        ]);
98        offset += 4;
99
100        let sighash;
101        let orchard_alphas;
102
103        if shielding {
104            // shielding format: multiple transparent inputs, no orchard alphas
105            // [input_count: 2B LE][per-input: sighash(32B) + address_index(4B LE)]...[action_count=0: 2B LE]
106            if offset + 2 > data.len() {
107                return Err(Error::ZcashParseError("input count truncated".to_string()));
108            }
109            let input_count = u16::from_le_bytes([data[offset], data[offset + 1]]) as usize;
110            offset += 2;
111
112            // use the first input's sighash as the representative sighash
113            if input_count == 0 {
114                return Err(Error::ZcashParseError("shielding: no inputs".to_string()));
115            }
116
117            let mut first_sighash = [0u8; 32];
118            // parse all inputs (sighash + address_index per input)
119            for i in 0..input_count {
120                if offset + 36 > data.len() {
121                    return Err(Error::ZcashParseError(format!(
122                        "shielding input {} truncated",
123                        i
124                    )));
125                }
126                if i == 0 {
127                    first_sighash.copy_from_slice(&data[offset..offset + 32]);
128                }
129                offset += 32; // sighash
130                offset += 4; // address_index
131            }
132            sighash = first_sighash;
133
134            // action count (should be 0 for shielding)
135            if offset + 2 > data.len() {
136                return Err(Error::ZcashParseError("action count truncated".to_string()));
137            }
138            offset += 2; // skip action_count (0)
139            orchard_alphas = Vec::new();
140        } else {
141            // regular send format: single sighash + orchard alphas
142            if offset + 32 > data.len() {
143                return Err(Error::ZcashParseError("sighash truncated".to_string()));
144            }
145            let mut sh = [0u8; 32];
146            sh.copy_from_slice(&data[offset..offset + 32]);
147            sighash = sh;
148            offset += 32;
149
150            // action count
151            if offset + 2 > data.len() {
152                return Err(Error::ZcashParseError("action count truncated".to_string()));
153            }
154            let action_count = u16::from_le_bytes([data[offset], data[offset + 1]]) as usize;
155            offset += 2;
156
157            // each action's alpha (32 bytes each)
158            let mut alphas = Vec::with_capacity(action_count);
159            for _ in 0..action_count {
160                if offset + 32 > data.len() {
161                    return Err(Error::ZcashParseError("alpha truncated".to_string()));
162                }
163                let mut alpha = [0u8; 32];
164                alpha.copy_from_slice(&data[offset..offset + 32]);
165                alphas.push(alpha);
166                offset += 32;
167            }
168            orchard_alphas = alphas;
169        }
170
171        // summary (length-prefixed string)
172        let summary = if offset + 2 <= data.len() {
173            let summary_len = u16::from_le_bytes([data[offset], data[offset + 1]]) as usize;
174            offset += 2;
175            if offset + summary_len <= data.len() {
176                String::from_utf8_lossy(&data[offset..offset + summary_len]).to_string()
177            } else {
178                String::new()
179            }
180        } else {
181            String::new()
182        };
183
184        Ok(Self {
185            account_index,
186            sighash,
187            orchard_alphas,
188            summary,
189            mainnet,
190            shielding,
191        })
192    }
193}

195/// create transaction cards for display
196fn create_zcash_cards(request: &ZcashSignRequestData) -> TransactionCardSet {
197    let mut method_cards = Vec::new();
198    let mut index = 0u32;
199
200    // add summary card
201    method_cards.push(TransactionCard {
202        index,
203        indent: 0,
204        card: Card::ZcashSummaryCard {
205            f: ZcashTransactionSummary {
206                mainnet: request.mainnet,
207                fee: "see summary".to_string(),
208                spend_count: request.orchard_alphas.len() as u64,
209                output_count: 0, // not provided in sign request
210                sighash: hex::encode(request.sighash),
211                anchor: "provided by wallet".to_string(),
212            },
213        },
214    });
215    index += 1;
216
217    // add spend cards for each action
218    for (i, alpha) in request.orchard_alphas.iter().enumerate() {
219        method_cards.push(TransactionCard {
220            index,
221            indent: 0,
222            card: Card::ZcashOrchardSpendCard {
223                f: ZcashOrchardSpend {
224                    nullifier: format!("action #{}", i + 1),
225                    value: "shielded".to_string(),
226                    cmx: hex::encode(&alpha[..16]), // show part of alpha as identifier
227                },
228            },
229        });
230        index += 1;
231    }
232
233    // add transaction summary from wallet if provided
234    if !request.summary.is_empty() {
235        // parse the summary to extract more details
236        // format is typically: "Send X.XX ZEC to u1..."
237        method_cards.push(TransactionCard {
238            index,
239            indent: 0,
240            card: Card::TextCard {
241                f: request.summary.clone(),
242            },
243        });
244        index += 1;
245    }
246
247    // add signing info
248    method_cards.push(TransactionCard {
249        index,
250        indent: 0,
251        card: Card::TextCard {
252            f: format!(
253                "Account #{} · {} signature{} required",
254                request.account_index,
255                request.orchard_alphas.len(),
256                if request.orchard_alphas.len() == 1 {
257                    ""
258                } else {
259                    "s"
260                }
261            ),
262        },
263    });
264
265    TransactionCardSet {
266        method: Some(method_cards),
267        ..Default::default()
268    }
269}

271/// process a zcash sign request QR code
272///
273/// this is called from handle_scanner_input when crypto type is "04" and tx type is "02"
274pub fn process_zcash_sign_request(
275    _database: &sled::Db,
276    data_hex: &str,
277) -> Result<TransactionAction> {
278    // parse the sign request
279    let request = ZcashSignRequestData::from_qr_hex(data_hex)?;
280
281    // create display cards
282    let content = create_zcash_cards(&request);
283
284    // return as Sign action - the signing will happen when user approves
285    // for now we return Read since we need to wire up the full signing flow
286    Ok(TransactionAction::Read {
287        r: Box::new(content),
288    })
289}

291/// check if a payload is a zcash transaction
292pub fn is_zcash_transaction(data_hex: &str) -> bool {
293    if data_hex.len() < 6 {
294        return false;
295    }
296    // check for 53 04 02 prefix
297    &data_hex[0..2] == "53" && &data_hex[2..4] == "04" && &data_hex[4..6] == "02"
298}

305    #[test]
306    fn test_parse_zcash_sign_request() {
307        // build a sign request manually
308        let mut data = Vec::new();
309
310        // prelude
311        data.push(0x53);
312        data.push(0x04);
313        data.push(0x02);
314
315        // flags: mainnet = true
316        data.push(0x01);
317
318        // account index
319        data.extend_from_slice(&0u32.to_le_bytes());
320
321        // sighash
322        let sighash = [0x42u8; 32];
323        data.extend_from_slice(&sighash);
324
325        // action count
326        data.extend_from_slice(&2u16.to_le_bytes());
327
328        // alpha values
329        let alpha1 = [0x11u8; 32];
330        let alpha2 = [0x22u8; 32];
331        data.extend_from_slice(&alpha1);
332        data.extend_from_slice(&alpha2);
333
334        // summary
335        let summary = "Send 1.5 ZEC";
336        data.extend_from_slice(&(summary.len() as u16).to_le_bytes());
337        data.extend_from_slice(summary.as_bytes());
338
339        // parse
340        let hex_str = hex::encode(&data);
341        let request = ZcashSignRequestData::from_qr_hex(&hex_str).unwrap();
342
343        assert_eq!(request.account_index, 0);
344        assert_eq!(request.sighash, sighash);
345        assert_eq!(request.orchard_alphas.len(), 2);
346        assert_eq!(request.orchard_alphas[0], alpha1);
347        assert_eq!(request.orchard_alphas[1], alpha2);
348        assert_eq!(request.summary, summary);
349        assert!(request.mainnet);
350    }
