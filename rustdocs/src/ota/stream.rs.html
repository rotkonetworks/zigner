1//! Stream verification: the device's load-bearing gate (§6.2).
2//!
3//! Input is the reassembled `ur:zafu-stream` buffer = `manifest_cbor ||
4//! image_wrapper` (manifest is frame[0], at byte offset 0). Validates the
5//! manifest signature + image signature against the pinned key, then the
6//! applicability gates (blacklist / board / class / anti-rollback), then
7//! decodes the trailing image wrapper and re-derives the full-image hash.
8
9use sha2::{Digest, Sha256};
10
11use crate::semver;
12use crate::types::{decode_manifest, ImageHeader, Manifest, CLASS_RELEASE, CLASS_ROLLBACK};
13use crate::verifysig::{verify_image, verify_manifest};
14use crate::{decode_one, Error};
15
16/// Fail with a formatted crate error (internal helper).
17macro_rules! err {
18    ($($t:tt)*) => { return Err(Error(format!($($t)*))) };
19}
20
21/// Result of a fully verified stream (data eligible for verified staging).
22#[derive(Debug, Clone)]
23pub struct VerifiedStream {
24    pub manifest: Manifest,
25    pub image_header: ImageHeader,
26    pub payload_sha256: [u8; 32],
27    pub payload_len: u64,
28}
29
30/// Default staging chunk size used for per-chunk hash verification.
31pub const STAGING_CHUNK: usize = 256;
32
33/// v1 module-image size cap (bytes). The delivered OTA target is a small
34/// signed module image, not a whole-firmware replacement; the freeze spec's
35/// open item suggests ~250 KB for v1 but the product frame is a few MB.
36/// We cap at 3 MiB so a QR scan stays ~1-2 minutes and a runaway/malformed
37/// stream can never claim a huge payload to stage.
38pub const MAX_MODULE_SIZE: u64 = 3 * 1024 * 1024;
39
40/// Verify a full `ur:zafu-stream` buffer.
41///
42/// * `payload_bytes`  — reassembled buffer: `manifest_cbor || image_wrapper`.
43/// * `manifest_sig`   — 64-byte manifest signature over fields 1..7. The
44///   frozen wire carries the image_sig (manifest field 8) inline but does NOT
45///   embed the manifest signature; the scan/carrier layer provides it.
46/// * `pinned_pubkey`  — 32-byte Zafu update public key (burned in).
47/// * `current_version`— currently applied firmware version (anti-rollback).
48/// * `board`          — expected target board name.
49/// * `blacklist`      — revoked `key_id`s (empty in dev).
50///
51/// `rollback_on_wire` controls whether a `rollback`-class stream is accepted
52/// here. The wire/apply path MUST keep it `false` (rollback is a device-local,
53/// human-gated action, never applied from a stream).
54pub fn verify_stream(
55    payload_bytes: &[u8],
56    manifest_sig: &[u8; 64],
57    pinned_pubkey: &[u8; 32],
58    current_version: &str,
59    board: &str,
60    blacklist: &[u64],
61    rollback_on_wire: bool,
62) -> Result<VerifiedStream, Error> {
63    // ── 1. Parse manifest at offset 0 (self-delimiting CBOR) ──────────────
64    if payload_bytes.is_empty() {
65        err!("empty stream buffer");
66    }
67    let (manifest, manifest_len) = decode_manifest_prefix(payload_bytes)?;
68    let image_sig = manifest
69        .image_sig
70        .as_ref()
71        .ok_or_else(|| Error("manifest has no image_sig on the wire".into()))?;
72    let _ = manifest
73        .req_id
74        .as_ref()
75        .ok_or_else(|| Error("manifest has no req_id on the wire".into()))?;
76
77    // Size discipline: the OTA target is a small signed module image.
78    if manifest.payload_size > MAX_MODULE_SIZE {
79        err!(
80            "payload_size {} exceeds the {} byte module-image cap",
81            manifest.payload_size,
82            MAX_MODULE_SIZE
83        );
84    }
85
86    // ── 2. Signature verification against pinned key ──────────────────────
87    verify_manifest(&manifest, manifest_sig, pinned_pubkey)?;
88
89    // Build the signed image header from the manifest field set.
90    let header = ImageHeader {
91        key_id: manifest.key_id,
92        board: manifest.board.clone(),
93        version: manifest.version.clone(),
94        payload_sha256: manifest.payload_sha256,
95        payload_len: manifest.payload_size,
96    };
97    verify_image(&header, image_sig, pinned_pubkey)?;
98
99    // ── 3. Applicability gates ────────────────────────────────────────────
100    if blacklist.contains(&manifest.key_id) {
101        err!("key_id {} is blacklisted/revoked", manifest.key_id);
102    }
103
104    if manifest.board != board {
105        err!(
106            "board mismatch: stream targets '{}' but device is '{}'",
107            manifest.board,
108            board
109        );
110    }
111
112    match manifest.class.as_str() {
113        CLASS_RELEASE => {}
114        CLASS_ROLLBACK => {
115            if !rollback_on_wire {
116                err!("rollback-class stream is not applicable on the wire (device-local only)");
117            }
118        }
119        other => err!("unknown firmware class: {other}"),
120    }
121
122    // ── 4. Anti-rollback / downgrade (normative semver) ───────────────────
123    let version = semver::parse(&manifest.version)
124        .ok_or_else(|| Error(format!("invalid firmware version '{}'", manifest.version)))?;
125    let min_version = semver::parse(&manifest.min_version)
126        .ok_or_else(|| Error(format!("invalid min_version '{}'", manifest.min_version)))?;
127    if version < min_version {
128        err!(
129            "version {} is below the signed min_version {}",
130            manifest.version,
131            manifest.min_version
132        );
133    }
134    if let Some(cur) = semver::parse(current_version) {
135        if version <= cur {
136            err!(
137                "version {} is not > current {} (anti-rollback)",
138                manifest.version,
139                current_version
140            );
141        }
142    } else {
143        // current unknown: fail closed rather than allow a possibly-older image.
144        err!("cannot establish current firmware version '{current_version}' for anti-rollback");
145    }
146
147    // ── 5. Decode trailing image wrapper ──────────────────────────────────
148    let wrapper = &payload_bytes[manifest_len..];
149    let (wrapper_key_id, wrapper_payload_len, wrapper_sha, payload) =
150        decode_image_wrapper(wrapper)?;
151
152    if wrapper_key_id != manifest.key_id {
153        err!("image wrapper key_id mismatch");
154    }
155    if wrapper_payload_len != manifest.payload_size {
156        err!("image wrapper payload_len mismatch with manifest payload_size");
157    }
158    if wrapper_sha != manifest.payload_sha256 {
159        err!("image wrapper payload sha256 mismatch with manifest");
160    }
161    if payload.len() as u64 != wrapper_payload_len {
162        err!("image wrapper payload length mismatch");
163    }
164
165    verify_payload_chunks(payload, &manifest.payload_sha256, STAGING_CHUNK)?;
166
167    Ok(VerifiedStream {
168        manifest: manifest.clone(),
169        image_header: header,
170        payload_sha256: manifest.payload_sha256,
171        payload_len: manifest.payload_size,
172    })
173}

175/// Decode the manifest at offset 0 and return (manifest, consumed_bytes).
176fn decode_manifest_prefix(data: &[u8]) -> Result<(Manifest, usize), Error> {
177    let (_cbor, consumed) = decode_one(data)?;
178    let bytes = &data[..consumed];
179    decode_manifest(bytes).map(|m| (m, consumed))
180}

182/// Parse `u32le key_id || u32le payload_len || sha256(32) || payload`.
183/// `(chunk_index, total_chunks, image_hash, chunk_bytes)` of one streamed
184/// image wrapper.
185pub type ImageWrapper<'a> = (u64, u64, [u8; 32], &'a [u8]);
186
187pub fn decode_image_wrapper(wrapper: &[u8]) -> Result<ImageWrapper<'_>, Error> {
188    if wrapper.len() < 8 + 32 {
189        err!("image wrapper too short");
190    }
191    let key_id = u32::from_le_bytes(wrapper[0..4].try_into().unwrap()) as u64;
192    let payload_len = u32::from_le_bytes(wrapper[4..8].try_into().unwrap()) as u64;
193    let mut sha = [0u8; 32];
194    sha.copy_from_slice(&wrapper[8..40]);
195    let payload = &wrapper[40..];
196    if payload.len() as u64 != payload_len {
197        err!(
198            "image wrapper payload_len {} != actual {}",
199            payload_len,
200            payload.len()
201        );
202    }
203    Ok((key_id, payload_len, sha, payload))
204}

206/// Verify the full-image sha256 by iterating fixed chunks — the per-chunk
207/// staging loop. Each chunk is individually hashed so a staged write can be
208/// re-verified; the concatenated hash must equal `expected`.
209pub fn verify_payload_chunks(
210    payload: &[u8],
211    expected: &[u8; 32],
212    chunk_size: usize,
213) -> Result<(), Error> {
214    if chunk_size == 0 {
215        err!("chunk size must be > 0");
216    }
217    let mut hasher = Sha256::new();
218    let mut n = 0usize;
219    while n < payload.len() {
220        let end = (n + chunk_size).min(payload.len());
221        let chunk = &payload[n..end];
222        let mut chunk_hash = Sha256::new();
223        chunk_hash.update(chunk);
224        let _ = chunk_hash.finalize();
225        hasher.update(chunk);
226        n = end;
227    }
228    let digest: [u8; 32] = hasher.finalize().into();
229    if digest != *expected {
230        err!(
231            "payload sha256 mismatch: computed {} expected {}",
232            hex::encode(digest),
233            hex::encode(expected)
234        );
235    }
236    Ok(())
237}
