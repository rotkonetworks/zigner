1//! Signed module packages: the trust layer of the update architecture.
2//!
3//! package = manifest || module_bytes
4//! manifest (fixed order, little-endian):
5//!   magic "ZIGM" | manifest_version:u8=1 | module_version:u32 |
6//!   min_kernel_version:u32 | module_hash:[32] | desc_len:u8 | desc |
7//!   sig_count:u8 | sig_count * ( key_index:u8 | sig:[64] )
8//!
9//! Signatures cover every manifest byte before sig_count over the
10//! domain-separated message "zigner-module-v1" || fields. 2-of-3 of the
11//! kernel-baked release keys must verify, each from a DISTINCT key.
12//! module_hash must equal sha256(module_bytes). The kernel additionally
13//! enforces module_version > last_installed (anti-rollback) - persisted
14//! counter, outside this parser.
15
16use ed25519_dalek::{Signature, Verifier, VerifyingKey};
17use sha2::{Digest, Sha256};
18
19pub const MAGIC: &[u8; 4] = b"ZIGM";
20pub const DOMAIN: &[u8] = b"zigner-module-v1";
21pub const REQUIRED_SIGS: usize = 2;
22
23#[derive(Debug)]
24pub enum ManifestError {
25    Truncated,
26    BadMagic,
27    UnsupportedVersion(u8),
28    HashMismatch,
29    DuplicateKey(u8),
30    UnknownKey(u8),
31    BadSignature(u8),
32    NotEnoughSignatures(usize),
33    Rollback { offered: u32, installed: u32 },
34    KernelTooOld { required: u32, running: u32 },
35}
36
37pub struct VerifiedModule<'a> {
38    pub module_version: u32,
39    pub min_kernel_version: u32,
40    pub description: String,
41    pub module_bytes: &'a [u8],
42}
43
44/// Parse + fully verify a module package against the kernel's trust
45/// anchors and state. Returns the module only if EVERY check passes.
46pub fn verify_package<'a>(
47    package: &'a [u8],
48    release_keys: &[VerifyingKey; 3],
49    kernel_version: u32,
50    last_installed_version: u32,
51) -> Result<VerifiedModule<'a>, ManifestError> {
52    let take = |b: &'a [u8], n: usize| -> Result<(&'a [u8], &'a [u8]), ManifestError> {
53        if b.len() < n {
54            Err(ManifestError::Truncated)
55        } else {
56            Ok(b.split_at(n))
57        }
58    };
59
60    let (magic, rest) = take(package, 4)?;
61    if magic != MAGIC {
62        return Err(ManifestError::BadMagic);
63    }
64    let (ver, rest) = take(rest, 1)?;
65    if ver[0] != 1 {
66        return Err(ManifestError::UnsupportedVersion(ver[0]));
67    }
68    let (mv, rest) = take(rest, 4)?;
69    let module_version = u32::from_le_bytes(mv.try_into().unwrap());
70    let (kv, rest) = take(rest, 4)?;
71    let min_kernel_version = u32::from_le_bytes(kv.try_into().unwrap());
72    let (hash, rest) = take(rest, 32)?;
73    let (dl, rest) = take(rest, 1)?;
74    let (desc, rest) = take(rest, dl[0] as usize)?;
75
76    // signed region = everything up to here
77    let signed_len = package.len() - rest.len();
78    let mut msg = Vec::with_capacity(DOMAIN.len() + signed_len);
79    msg.extend_from_slice(DOMAIN);
80    msg.extend_from_slice(&package[..signed_len]);
81
82    let (sc, mut rest) = take(rest, 1)?;
83    let mut seen = [false; 3];
84    let mut valid = 0usize;
85    for _ in 0..sc[0] {
86        let (ki, r) = take(rest, 1)?;
87        let (sig, r) = take(r, 64)?;
88        rest = r;
89        let idx = ki[0] as usize;
90        if idx >= 3 {
91            return Err(ManifestError::UnknownKey(ki[0]));
92        }
93        if seen[idx] {
94            return Err(ManifestError::DuplicateKey(ki[0]));
95        }
96        seen[idx] = true;
97        let signature = Signature::from_bytes(sig.try_into().unwrap());
98        release_keys[idx]
99            .verify(&msg, &signature)
100            .map_err(|_| ManifestError::BadSignature(ki[0]))?;
101        valid += 1;
102    }
103    if valid < REQUIRED_SIGS {
104        return Err(ManifestError::NotEnoughSignatures(valid));
105    }
106
107    let module_bytes = rest;
108    if <[u8; 32]>::from(Sha256::digest(module_bytes)) != *<&[u8; 32]>::try_from(hash).unwrap() {
109        return Err(ManifestError::HashMismatch);
110    }
111    if module_version <= last_installed_version {
112        return Err(ManifestError::Rollback {
113            offered: module_version,
114            installed: last_installed_version,
115        });
116    }
117    if min_kernel_version > kernel_version {
118        return Err(ManifestError::KernelTooOld {
119            required: min_kernel_version,
120            running: kernel_version,
121        });
122    }
123
124    Ok(VerifiedModule {
125        module_version,
126        min_kernel_version,
127        description: String::from_utf8_lossy(desc).into_owned(),
128        module_bytes,
129    })
130}

132/// Packaging helper (release tooling + tests): sign with the provided keys.
133pub fn build_package(
134    module_bytes: &[u8],
135    module_version: u32,
136    min_kernel_version: u32,
137    description: &str,
138    signers: &[(u8, ed25519_dalek::SigningKey)],
139) -> Vec<u8> {
140    use ed25519_dalek::Signer as _;
141    let mut out = Vec::new();
142    out.extend_from_slice(MAGIC);
143    out.push(1);
144    out.extend_from_slice(&module_version.to_le_bytes());
145    out.extend_from_slice(&min_kernel_version.to_le_bytes());
146    out.extend_from_slice(&Sha256::digest(module_bytes));
147    out.push(description.len() as u8);
148    out.extend_from_slice(description.as_bytes());
149    let mut msg = Vec::from(DOMAIN);
150    msg.extend_from_slice(&out);
151    out.push(signers.len() as u8);
152    for (idx, key) in signers {
153        out.push(*idx);
154        out.extend_from_slice(&key.sign(&msg).to_bytes());
155    }
156    out.extend_from_slice(module_bytes);
157    out
158}
