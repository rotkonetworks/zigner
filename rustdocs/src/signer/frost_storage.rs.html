1//! FROST key shares at rest.
2//!
3//! A FROST share has no seed phrase behind it: it exists only in this
4//! device's database, so it used to sit there as plaintext JSON. It is now
5//! sealed with XChaCha20-Poly1305 under a key derived from one of the
6//! device's seeds, the seed the user already backs up:
7//!
8//! ```text
9//!   key  = HMAC-SHA512("zigner-frost-storage-v1", seed)[..32]
10//!   aad  = "zigner-frost-v1" || wallet_id
11//!   blob = version(1) || nonce(24) || ciphertext+tag
12//! ```
13//!
14//! The wallet id in the associated data stops one wallet's sealed secrets
15//! being moved onto another wallet's row. The record keeps a fingerprint of
16//! the sealing seed so the right seed is found on a device with several.
17//!
18//! Reading a share therefore needs the seed, which needs the unlock prompt.
19//! A wallet stored before sealing is sealed the first time its secrets are
20//! read with a seed at hand, and only after the sealed copy has been written
21//! and read back.
22
23use chacha20poly1305::{
24    aead::{Aead, KeyInit, Payload},
25    XChaCha20Poly1305, XNonce,
26};
27use db_handling::frost::FrostWalletData;
28use hmac::{Hmac, Mac};
29use sha2::Sha512;
30use zeroize::Zeroizing;
31
32const KEY_DOMAIN: &[u8] = b"zigner-frost-storage-v1";
33const ID_DOMAIN: &[u8] = b"zigner-frost-storage-id-v1";
34const AAD_DOMAIN: &[u8] = b"zigner-frost-v1";
35const SEALED_V1: u8 = 1;
36
37/// A wallet's two secrets, wiped on drop.
38pub struct Secrets {
39    pub key_package_hex: Zeroizing<String>,
40    pub ephemeral_seed_hex: Zeroizing<String>,
41}
42
43fn hmac(key: &[u8], data: &[u8]) -> Zeroizing<[u8; 64]> {
44    let mut mac = <Hmac<Sha512> as Mac>::new_from_slice(key).expect("HMAC takes any key size");
45    mac.update(data);
46    Zeroizing::new(mac.finalize().into_bytes().into())
47}
48
49fn storage_key(seed_phrase: &str) -> Zeroizing<[u8; 32]> {
50    let wide = hmac(KEY_DOMAIN, seed_phrase.as_bytes());
51    let mut key = Zeroizing::new([0u8; 32]);
52    key.copy_from_slice(&wide[..32]);
53    key
54}
55
56/// Short public fingerprint of a seed, stored beside the sealed secrets.
57pub fn seed_id(seed_phrase: &str) -> String {
58    hex::encode(&hmac(ID_DOMAIN, seed_phrase.as_bytes())[..8])
59}
60
61fn aad(wallet_id: &str) -> Vec<u8> {
62    [AAD_DOMAIN, wallet_id.as_bytes()].concat()
63}
64
65fn seal(seed_phrase: &str, wallet_id: &str, secrets: &Secrets) -> Result<String, String> {
66    let plaintext = Zeroizing::new(
67        serde_json::to_vec(&serde_json::json!({
68            "k": secrets.key_package_hex.as_str(),
69            "e": secrets.ephemeral_seed_hex.as_str(),
70        }))
71        .map_err(|e| e.to_string())?,
72    );
73    let mut nonce = [0u8; 24];
74    crate::backup::getrandom(&mut nonce)?;
75    let cipher = XChaCha20Poly1305::new((&*storage_key(seed_phrase)).into());
76    let ct = cipher
77        .encrypt(
78            XNonce::from_slice(&nonce),
79            Payload {
80                msg: &plaintext,
81                aad: &aad(wallet_id),
82            },
83        )
84        .map_err(|_| "sealing the FROST share failed".to_string())?;
85    let mut blob = Vec::with_capacity(1 + 24 + ct.len());
86    blob.push(SEALED_V1);
87    blob.extend_from_slice(&nonce);
88    blob.extend_from_slice(&ct);
89    Ok(hex::encode(blob))
90}
91
92fn open(seed_phrase: &str, wallet_id: &str, sealed_hex: &str) -> Result<Secrets, String> {
93    let blob = hex::decode(sealed_hex).map_err(|_| "sealed FROST share is not hex")?;
94    if blob.len() < 1 + 24 + 16 || blob[0] != SEALED_V1 {
95        return Err("sealed FROST share has an unknown format".into());
96    }
97    let cipher = XChaCha20Poly1305::new((&*storage_key(seed_phrase)).into());
98    let plaintext = Zeroizing::new(
99        cipher
100            .decrypt(
101                XNonce::from_slice(&blob[1..25]),
102                Payload {
103                    msg: &blob[25..],
104                    aad: &aad(wallet_id),
105                },
106            )
107            .map_err(|_| "the FROST share does not open with this key set".to_string())?,
108    );
109    let v: serde_json::Value = serde_json::from_slice(&plaintext).map_err(|e| e.to_string())?;
110    let field = |k: &str| {
111        v[k].as_str()
112            .map(|s| Zeroizing::new(s.to_string()))
113            .ok_or_else(|| format!("sealed FROST share lacks `{k}`"))
114    };
115    Ok(Secrets {
116        key_package_hex: field("k")?,
117        ephemeral_seed_hex: field("e")?,
118    })
119}

121/// Seal `data`'s secrets in place under `seed_phrase`, clearing the plaintext
122/// fields. The result is checked by opening it again before it is returned.
123pub fn seal_wallet(data: &mut FrostWalletData, seed_phrase: &str) -> Result<(), String> {
124    let wallet_id = db_handling::frost::wallet_id_hex(&data.public_key_package_hex);
125    let secrets = Secrets {
126        key_package_hex: Zeroizing::new(std::mem::take(&mut data.key_package_hex)),
127        ephemeral_seed_hex: Zeroizing::new(std::mem::take(&mut data.ephemeral_seed_hex)),
128    };
129    let sealed = seal(seed_phrase, &wallet_id, &secrets)?;
130    let back = open(seed_phrase, &wallet_id, &sealed)?;
131    if *back.key_package_hex != *secrets.key_package_hex
132        || *back.ephemeral_seed_hex != *secrets.ephemeral_seed_hex
133    {
134        return Err("sealed FROST share did not read back".into());
135    }
136    data.sealed = Some(sealed);
137    data.seed_id = Some(seed_id(seed_phrase));
138    Ok(())
139}

141/// The secrets of a stored wallet.
142///
143/// A sealed wallet opens with whichever of `seed_phrases` it was sealed
144/// under. A wallet stored before sealing is returned as is and, when a seed
145/// is available, sealed under the first one and written back - only after
146/// the sealed copy opens to the same secrets, so a failure leaves the old
147/// record untouched.
148pub fn wallet_secrets(
149    database: &sled::Db,
150    data: &FrostWalletData,
151    seed_phrases: &[String],
152) -> Result<Secrets, String> {
153    let wallet_id = db_handling::frost::wallet_id_hex(&data.public_key_package_hex);
154    if let Some(sealed) = &data.sealed {
155        let seed = seed_phrases
156            .iter()
157            .find(|s| Some(seed_id(s)) == data.seed_id)
158            .ok_or("the key set this FROST wallet is sealed under is not on this device")?;
159        return open(seed, &wallet_id, sealed);
160    }
161
162    let secrets = Secrets {
163        key_package_hex: Zeroizing::new(data.key_package_hex.clone()),
164        ephemeral_seed_hex: Zeroizing::new(data.ephemeral_seed_hex.clone()),
165    };
166    if let Some(seed) = seed_phrases.first() {
167        let mut sealed = data.clone();
168        seal_wallet(&mut sealed, seed)?;
169        db_handling::frost::store_frost_wallet(database, &sealed)
170            .map_err(|e| format!("sealing the stored FROST share: {e}"))?;
171        let stored = db_handling::frost::get_frost_wallet(database, &wallet_id)
172            .map_err(|e| e.to_string())?
173            .ok_or("sealed FROST wallet vanished")?;
174        let back = wallet_secrets(database, &stored, seed_phrases)?;
175        if *back.key_package_hex != *secrets.key_package_hex {
176            return Err("sealed FROST share did not read back".into());
177        }
178    }
179    Ok(secrets)
180}

182#[cfg(test)]
183mod tests {
184    use super::*;
185
186    const SEED_A: &str = "abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon about";
187    const SEED_B: &str =
188        "legal winner thank year wave sausage worth useful legal winner thank yellow";
189
190    fn plaintext_wallet(pkg: &str) -> FrostWalletData {
191        FrostWalletData {
192            key_package_hex: "kp-secret".into(),
193            public_key_package_hex: pkg.into(),
194            ephemeral_seed_hex: "es-secret".into(),
195            label: "2-of-3".into(),
196            min_signers: 2,
197            max_signers: 3,
198            mainnet: true,
199            created_at: 0,
200            orchard_fvk_uview: None,
201            address: None,
202            relay_url: None,
203            sealed: None,
204            seed_id: None,
205        }
206    }
207
208    #[test]
209    fn sealed_wallet_holds_no_plaintext_and_opens_with_its_seed() {
210        let db = sled::Config::new().temporary(true).open().unwrap();
211        let mut w = plaintext_wallet("aa01");
212        seal_wallet(&mut w, SEED_A).unwrap();
213        assert!(w.key_package_hex.is_empty() && w.ephemeral_seed_hex.is_empty());
214        let raw = serde_json::to_string(&w).unwrap();
215        assert!(!raw.contains("kp-secret") && !raw.contains("es-secret"));
216
217        db_handling::frost::store_frost_wallet(&db, &w).unwrap();
218        let seeds = vec![SEED_B.to_string(), SEED_A.to_string()];
219        let s = wallet_secrets(&db, &w, &seeds).unwrap();
220        assert_eq!(*s.key_package_hex, "kp-secret");
221        assert_eq!(*s.ephemeral_seed_hex, "es-secret");
222    }
223
224    #[test]
225    fn wrong_seed_or_moved_blob_is_refused_without_change() {
226        let db = sled::Config::new().temporary(true).open().unwrap();
227        let mut w = plaintext_wallet("aa02");
228        seal_wallet(&mut w, SEED_A).unwrap();
229        assert!(wallet_secrets(&db, &w, &[SEED_B.to_string()]).is_err());
230
231        // the same blob on another wallet's record does not open (AAD)
232        let mut other = plaintext_wallet("bb02");
233        other.sealed = w.sealed.clone();
234        other.seed_id = w.seed_id.clone();
235        other.key_package_hex.clear();
236        other.ephemeral_seed_hex.clear();
237        assert!(wallet_secrets(&db, &other, &[SEED_A.to_string()]).is_err());
238    }
239
240    #[test]
241    fn plaintext_wallet_is_sealed_on_first_read() {
242        let db = sled::Config::new().temporary(true).open().unwrap();
243        let w = plaintext_wallet("aa03");
244        let id = db_handling::frost::store_frost_wallet(&db, &w).unwrap();
245
246        // no seed: works as before, record untouched
247        let s = wallet_secrets(&db, &w, &[]).unwrap();
248        assert_eq!(*s.key_package_hex, "kp-secret");
249        assert!(db_handling::frost::get_frost_wallet(&db, &id)
250            .unwrap()
251            .unwrap()
252            .sealed
253            .is_none());
254
255        // with a seed: same secrets back, and the record is now sealed
256        let s = wallet_secrets(&db, &w, &[SEED_A.to_string()]).unwrap();
257        assert_eq!(*s.ephemeral_seed_hex, "es-secret");
258        let stored = db_handling::frost::get_frost_wallet(&db, &id)
259            .unwrap()
260            .unwrap();
261        assert!(stored.sealed.is_some() && stored.key_package_hex.is_empty());
262        let again = wallet_secrets(&db, &stored, &[SEED_A.to_string()]).unwrap();
263        assert_eq!(*again.key_package_hex, "kp-secret");
264    }
